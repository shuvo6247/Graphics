    void renderBitmapString(float x, float y, float z, void *font, char *string) //rendering refers to the process of generating an image or a scene from a 3D model
    {
        char *c;
        glRasterPos3f(x, y,z); //sets the position in 3D space

        {
            glutBitmapCharacter(font, *c);//renders a single character from the string at the current raster position.
        }
    }

        void SpecialInput(int key, int x, int y)  // SpecialInput function to handle arrow key presses for speeding up or slowing down the vehicle
        {

            switch(key)
            {
            case GLUT_KEY_UP:
                if(vehicleSpeed >= 0) // Ensure vehicle doesn't stop if the speed is positive
                {
                    vehicleStop = false;
                }
                vehicleSpeed = vehicleSpeed + 0.001; // Increase speed
                break;

            case GLUT_KEY_DOWN:

                if(vehicleSpeed > 0)
                {
                    vehicleSpeed = vehicleSpeed - 0.001; // // Decrease speed
                }
                if(vehicleSpeed <= 0)
                {
                    vehicleStop = true; // // Stop vehicle if speed becomes zero or negative
                }
                break;
            }
            glutPostRedisplay();
        }

        // Input function to handle regular key presses (for toggling various settings)
        void Input(unsigned char key, int x, int y)
        {

            {
            case 'n':
                night = true;


                break;
            case 'd':
                night = false;

                break;

            case '0':
                on= false;
                break;

            case '1':
                on = true;
                break;

                case 's':
                        if(cover)
                {
                    cover = false;
                }
                else
                {
                    cover = true;
                }
                break;

            case '+':
                glutFullScreen();
                break;

            case '-':
                glutReshapeWindow(1024,576);
                glutInitWindowPosition(50,40);
                break;

            case 'x':

                exit(0);
                break;

            case 'r':
                rain = true;

                break;

            case 'e':

                rain = false;
                break;
            }
            glutPostRedisplay();
        }
