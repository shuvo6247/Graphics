void drawWave(float moveX, float moveY)
{
    glPushMatrix();

    glTranslatef(moveX + waveMove, moveY, 0);

    glColor3ub(220,240,255);
    glLineWidth(3);

    glBegin(GL_LINE_STRIP);

    int segments = 50;

    for(int i=0; i<=segments; i++)
    {
        float t = (float)i / segments;
        float omt = 1.0f - t;

        // Cubic Bézier formula
        float x = omt*omt*omt*waveX[0] + 3*t*omt*omt*waveX[1] + 3*t*t*omt*waveX[2] + t*t*t*waveX[3];
        float y = omt*omt*omt*waveY[0] + 3*t*omt*omt*waveY[1] + 3*t*t*omt*waveY[2] + t*t*t*waveY[3];

        glVertex2f(x, y);
    }

    glEnd();

    glPopMatrix();
}

void drawFog()
{
    if (!fogOn) return;

    glColor4f(0.90f, 0.92f, 0.95f, fogDensity * 0.75f);
    glBegin(GL_QUADS);
        glVertex2f(0,0);
        glVertex2f(1900,0);
        glVertex2f(1900,1000);
        glVertex2f(0,1000);
    glEnd();

    //Down
    float gndAlpha = fogDensity * 0.15f;
    glColor4f(0.93f, 0.94f, 0.97f, gndAlpha);
    glBegin(GL_QUADS);
        glVertex2f(0,0);
        glVertex2f(1900,0);
        glVertex2f(1900,280);
        glVertex2f(0,280);
    glEnd();

    float riverAlpha = fogDensity * 0.73f;
    glColor4f(0.88f, 0.90f, 0.94f, riverAlpha);
    glBegin(GL_QUADS);
        glVertex2f(0,280);
        glVertex2f(1900,280);
        glVertex2f(1900,650);
        glVertex2f(0,650);
    glEnd();

    // ---- Layer 4: Sky haze (top portion fades into grey-white) ----
    float skyAlpha = fogDensity * 1.25f;
    glColor4f(0.88f, 0.90f, 0.94f, skyAlpha);
    glBegin(GL_QUADS);
        glVertex2f(0,650);
        glVertex2f(1900,650);
        glVertex2f(1900,1000);
        glVertex2f(0,1000);
    glEnd();
}

void handleKeypress(unsigned char key, int x, int y)
{
    if (key == 'f' || key == 'F') {
        fogOn = !fogOn;
        glutPostRedisplay();
    }
    // fog density controls
    if (key == '+' && fogDensity < 0.9f) {
        fogDensity += 0.05f;
        glutPostRedisplay();
    }
    if (key == '-' && fogDensity > 0.1f) {
        fogDensity -= 0.05f;
        glutPostRedisplay();
    }

    switch(key)
    {
        // Train control
        case 't':
            train_speed += 1.0f;
            break;
        case 'g':
            train_speed -= 1.0f;
            break;
        case 's':
            train_speed = 0.0f;
            break;
        case 'r':
            train_speed = 7.0f;
            break;
    }

    switch(key)
    {
        //Car control
        case 'c':
            car2_speed += 1.0f;
            break;
        case 'v':
            car2_speed -= 1.0f;
            break;
        case 'b':
            car2_speed = 0.0f;
            break;
        case 'a':
            car2_speed = 6.0f;
            break;
    }
}

void handleMouse(int button, int state, int x, int y)
{
    if(button == GLUT_LEFT_BUTTON)
    {
        boatSpeed2 += 0.1f;
    }
    if (button == GLUT_RIGHT_BUTTON)
    {
        boatSpeed2 -= 0.1f;
    }
    glutPostRedisplay();
}


void init()
{
    glClearColor(0.5, 0.8, 1.0, 1.0);

    // Enable alpha blending globally (required for fog overlay)
    glEnable(GL_BLEND);
    glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);

    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();
    gluOrtho2D(0, 1900, 0, 1000);

    glMatrixMode(GL_MODELVIEW);
}

{

    glutInitWindowSize(1200, 700);

    init();

    glutDisplayFunc(display);

    return 0;
}
