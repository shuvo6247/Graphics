void Circle(float x, float y, float r)
{
    glBegin(GL_POLYGON);
    for (int i = 0; i < 500; i++)
    {
        float a = 2 * 3.1416f * i / 500;
        glVertex2f(x + cos(a) * r, y + sin(a) * r);
    }
    glEnd();
}

float moveWave[7] = {0,0,0,0,0,0,0};

// Cubic Bezier curve helper for smooth river waves
void BezierCurve(float x1, float y1, float x2, float y2,
                 float x3, float y3, float x4, float y4)
{
    glBegin(GL_LINE_STRIP);
    for(int i = 0; i <= 35; i++)
    {
        float t = i / 35.0f;
        float u = 1.0f - t;

        float x = (u*u*u*x1) + (3*u*u*t*x2) + (3*u*t*t*x3) + (t*t*t*x4);
        float y = (u*u*u*y1) + (3*u*u*t*y2) + (3*u*t*t*y3) + (t*t*t*y4);

        glVertex2f(x, y);
    }
    glEnd();
}

void DrawBezierWave(float x, float y, float width, float height)
{
    // First half: smooth upward curve
    BezierCurve(x, y,
                x + width * 0.18f, y + height,
                x + width * 0.32f, y + height,
                x + width * 0.50f, y);

    // Second half: smooth downward curve
    BezierCurve(x + width * 0.50f, y,
                x + width * 0.68f, y - height,
                x + width * 0.82f, y - height,
                x + width, y);
}

void MovingBezierWave(int index, float x, float y, float width, float height)
{
    glPushMatrix();
    glTranslatef(moveWave[index], 0, 0);

    DrawBezierWave(x, y, width, height);

    // Extra copy keeps the wave continuous while it wraps around the river
    DrawBezierWave(x - 400, y, width, height);

    glPopMatrix();
}

void moveCar(int value) // AF3
{
    moveCar1 += dirCar1 * speedCar1;
    moveCar2 += dirCar2 * speedCar2;

    // Move
    if(moveCar1 > 360)  moveCar1 = -360;
    if(moveCar1 < -360) moveCar1 = 360;

    if(moveCar2 > 360)  moveCar2 = -360;
    if(moveCar2 < -360) moveCar2 = 360;

    // brake and backlight
    if(speedCar1 == 0)
    blCar1 = 1;
    else
    blCar1 = 0;

    if(speedCar2 == 0)
    blCar2 = 1;
    else
    blCar2 = 0;

    // wheel speed sync
    wheelSpeed1 = speedCar1 * 1.5f;
    wheelSpeed2 = speedCar2 * 1.5f;

    glutPostRedisplay();
    glutTimerFunc(20, moveCar, 0);
}
void updateWheelRotation(int value) // AF4
{
    if(speedCar1 > 0)
        angle1 += dirCar1 * wheelSpeed1*(-1);

    if(speedCar2 > 0)
        angle2 += dirCar2 * wheelSpeed2*(-1);

    if(angle1 >= 360) angle1 -= 360;
    if(angle1 < 0)    angle1 += 360;

    if(angle2 >= 360) angle2 -= 360;
    if(angle2 < 0)    angle2 += 360;

    glutPostRedisplay();
    glutTimerFunc(16, updateWheelRotation, 0);
}

/////////////////////////////////////////////////////////////////////   HANDELKEY
void handleKeypress(unsigned char key, int x, int y)
{
    switch(key)
    {
        //////////////////////////////////////////////  Car 1

    case 'r':      // headlight
        hlCar1 = !hlCar1;
        break;

    case 'a':       // Reverse
        if(dirCar1 == 1)
            speedCar1 += accel;
        else
        {
            speedCar1 -= accel;
            if(speedCar1 <= 0)
            {
                speedCar1 = 0;
                dirCar1 = 1;
            }
        }
        break;

    case 'q':           // Forward
        if(dirCar1 == -1)
            speedCar1 += accel;
        else
        {
            speedCar1 -= accel;
            if(speedCar1 <= 0)
            {
                speedCar1 = 0;
                dirCar1 = -1;
            }
        }
        break;

    case 'z':           // stop
        speedCar1 = 0;
        wheelSpeed1 = 0;
        break;

    /////////////////////////////////////////////////  Car 2
    case 'f':           // headlight
        hlCar2 = !hlCar2;
        break;

    case 'w':
        if(dirCar2 == 1)
            speedCar2 += accel;
        else
        {
            speedCar2 -= accel;
            if(speedCar2 <= 0)
            {
                speedCar2 = 0;
                dirCar2 = 1;
            }
        }
        break;

    case 's':
        if(dirCar2 == -1)
            speedCar2 += accel;
        else
        {
            speedCar2 -= accel;
            if(speedCar2 <= 0)
            {
                speedCar2 = 0;
                dirCar2 = -1;
            }
        }
        break;

    case 'x':
        speedCar2 = 0;
        wheelSpeed2 = 0;
        break;

    case 'e':   // stars
        callStars = -1 * callStars;
        break;

    case 'd':   // Moon
        callMoon = -1 * callMoon;
        break;

    case 'c':   // leaf
        moveLeaf = -1 * moveLeaf;
        break;

    case 'v':   // nlRoad
        nlRoad = -1 * nlRoad;
        break;

    case '1':   // Building1
        nlBuilding1 = nlBuilding1 * -1;
        break;

    case '2':   // Building2
        nlBuilding2 = nlBuilding2 * -1;
        break;

    case '3':   // Building2
        nlBuilding3 = nlBuilding3 * -1;
        break;

    case '4':   // Building4
        nlBuilding4 = nlBuilding4 * -1;
        break;

    case '5':   // Building5
        nlBuilding5 = nlBuilding5 * -1;
        break;

    case '6':   // Building6
        nlBuilding6 = nlBuilding6 * -1;
        break;

    case '7':   // Building6
        nlBuilding7 = nlBuilding7 * -1;
        break;


    case 'y':   // BackBuilding1
        nlBackBuilding1 = nlBackBuilding1 * -1;
        break;

    case 'h':   // BackBuilding2
        nlBackBuilding2 = nlBackBuilding2 * -1;
        break;

    case 'n':   // BackBuilding2
        nlBackBuilding3 = nlBackBuilding3 * -1;
        break;

    case 'u':   // BackBuilding2
        nlBackBuilding4 = nlBackBuilding4 * -1;
        break;

    case 't':   // Boat
        moveBoat = moveBoat* -1;
        break;

    case 'g':   // Boat
        nlBoat = nlBoat* -1;
        break;

    case 'b':   // Evening and Night Seen
        nightSky = nightSky* -1;
        if(nightSky == -1)
        {
            nlRoad = -1;

            nlBuilding2 = -1;
            nlBuilding3 = -1;
            nlBuilding6 = -1;

            nlBackBuilding3 = -1;

            hlCar1 = -1;
            hlCar2 = -1;

            nlBoat = -1;

            ntWindow = -1;
        }
        else
        {
            nlRoad = 1;

            callStars = 1;

            nlBuilding1 = 1;
            nlBuilding2 = 1;
            nlBuilding3 = 1;
            nlBuilding4 = 1;
            nlBuilding5 = 1;
            nlBuilding6 = 1;
            nlBuilding7 = 1;

            nlBackBuilding1 = 1;
            nlBackBuilding2 = 1;
            nlBackBuilding3 = 1;
            nlBackBuilding4 = 1;

            hlCar1 = 1;
            hlCar2 = 1;

            nlBoat = 1;

            ntWindow = 1;
        }

        break;


    }

    // Max speed
    if(speedCar1 > maxSpeed) speedCar1 = maxSpeed;
    if(speedCar2 > maxSpeed) speedCar2 = maxSpeed;


}

void handleMouse(int button, int state, int x, int y)
{
    if (state == GLUT_DOWN)
    {
        // increase speed
        if (button == GLUT_LEFT_BUTTON)
        {
            moveClouds = 1;
        }

        // Stop clouds
        else if (button == GLUT_RIGHT_BUTTON)
        {
            moveClouds = -1;
            speedCloud = 0.0f;
        }

        //Faster clouds
        else if (button == 3)   // Scroll up
        {
            moveClouds = 1;
            speedCloud += 0.2f;
        }

        // Slower clouds
        else if (button == 4)   // Scroll down
        {
            speedCloud -= 0.2f;
        }
    }

    if(speedCloud > 5) speedCloud = 5;
    if(speedCloud < -5) speedCloud = -5;

    glutPostRedisplay();
}
